<script>
// ── State ─────────────────────────────────────────────
const WEBINARS = [
  {
    id: "w1",
    tag: "Webbinar 1",
    title: "Ny i Next",
    url: "https://helpdesk.next-tech.com/hc/sv/articles/24122929562524-Webinar-Ny-i-Next-2025-12-02"
  },
  {
    id: "w2",
    tag: "Webbinar 2",
    title: "Grundläggande ekonomiflöde i Next",
    url: "https://helpdesk.next-tech.com/hc/sv/articles/22232333563676-Webinar-Grundl%C3%A4ggande-ekonomifl%C3%B6de-i-Next-2025-09-09"
  },
  {
    id: "w3",
    tag: "Webbinar 3",
    title: "Budget, inköp och prognos",
    url: "https://helpdesk.next-tech.com/hc/sv/articles/26692106514716-Webinar-Budget-ink%C3%B6p-och-prognos-2026-04-09"
  }
];

const STORAGE_KEY = "bso_onboarding_" + encodeURIComponent(CONFIG.kundNamn);

let state = loadState();

function loadState() {
  try {
    const saved = localStorage.getItem(STORAGE_KEY);
    if (saved) return JSON.parse(saved);
  } catch(e) {}
  return {
    users: [
      { id: uid(), firstName: "", lastName: "", email: "" },
      { id: uid(), firstName: "", lastName: "", email: "" }
    ],
    webinarChecks: {},
    ansvarRead: false,
    giDone: false
  };
}

function saveState() {
  try { localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); } catch(e) {}
}

function uid() {
  return Math.random().toString(36).slice(2, 9);
}

// ── User management ────────────────────────────────────
function addUser() {
  state.users.push({ id: uid(), firstName: "", lastName: "", email: "" });
  saveState();
  renderAll();
}

function removeUser(id) {
  if (state.users.length <= 2) return;
  state.users = state.users.filter(u => u.id !== id);
  // Clean up webinar checks for this user
  WEBINARS.forEach(w => delete state.webinarChecks[id + "_" + w.id]);
  saveState();
  renderAll();
}

function updateUser(id, field, value) {
  const user = state.users.find(u => u.id === id);
  if (user) { user[field] = value; saveState(); updateProgress(); updateCta(); }
}

// ── Webinar checks ─────────────────────────────────────
function toggleWebinarCheck(userId, webinarId) {
  const key = userId + "_" + webinarId;
  state.webinarChecks[key] = !state.webinarChecks[key];
  saveState();
  renderAll();
}

function isWebinarChecked(userId, webinarId) {
  return !!state.webinarChecks[userId + "_" + webinarId];
}

// ── Single checks ──────────────────────────────────────
function toggleCheck(field) {
  state[field] = !state[field];
  saveState();
  renderAll();
}

// ── Validation helpers ─────────────────────────────────
function validUsers() {
  return state.users.filter(u =>
    u.firstName.trim() && u.lastName.trim() && u.email.trim() && u.email.includes("@")
  );
}

function usersReady() { return validUsers().length >= 2; }

function allWebinarsChecked() {
  if (!usersReady()) return false;
  const vUsers = validUsers();
  return WEBINARS.every(w => vUsers.every(u => isWebinarChecked(u.id, w.id)));
}

function allDone() {
  return usersReady() && allWebinarsChecked() && state.ansvarRead && state.giDone;
}

// ── Progress ───────────────────────────────────────────
function calcProgress() {
  const vUsers = validUsers();
  let total = 0, done = 0;

  // Users (counts as 1 chunk)
  total += 1;
  if (usersReady()) done += 1;

  // Webinars: 3 per valid user
  if (vUsers.length >= 2) {
    const wTotal = WEBINARS.length * vUsers.length;
    const wDone = WEBINARS.reduce((acc, w) => acc + vUsers.filter(u => isWebinarChecked(u.id, w.id)).length, 0);
    total += wTotal;
    done += wDone;
  } else {
    total += WEBINARS.length * 2; // assume min 2
  }

  // Ansvar + GI
  total += 2;
  if (state.ansvarRead) done += 1;
  if (state.giDone) done += 1;

  return Math.round((done / total) * 100);
}

function renderUsers() {
  const container = document.getElementById("userRows");
  container.innerHTML = "";
  state.users.forEach((user, idx) => {
    const row = document.createElement("div");
    row.className = "user-row";
    row.innerHTML = `
      <div class="input-group">
        ${idx === 0 ? '<span class="input-label">Förnamn</span>' : ''}
        <input type="text" placeholder="Förnamn" value="${esc(user.firstName)}"
          oninput="updateUser('${user.id}','firstName',this.value)" />
      </div>
      <div class="input-group">
        ${idx === 0 ? '<span class="input-label">Efternamn</span>' : ''}
        <input type="text" placeholder="Efternamn" value="${esc(user.lastName)}"
          oninput="updateUser('${user.id}','lastName',this.value)" />
      </div>
      <div class="input-group">
        ${idx === 0 ? '<span class="input-label">E-postadress</span>' : ''}
        <input type="email" placeholder="namn@foretag.se" value="${esc(user.email)}"
          oninput="updateUser('${user.id}','email',this.value)" />
      </div>
      <button class="btn-remove" onclick="removeUser('${user.id}')"
        ${state.users.length <= 2 ? 'disabled' : ''} title="Ta bort">✕</button>
    `;
    container.appendChild(row);
  });

  const note = document.getElementById("userMinNote");
  const vCount = validUsers().length;
  if (vCount >= 2) {
    note.textContent = `✓ ${vCount} superanvändare registrerade.`;
    note.className = "user-min-note";
  } else {
    note.textContent = `Minst 2 superanvändare med namn och e-post krävs. (${vCount}/2 klara)`;
    note.className = "user-min-note warning";
  }
}

function renderWebinars() {
  const container = document.getElementById("webinarList");
  container.innerHTML = "";
  const vUsers = validUsers();

  WEBINARS.forEach(w => {
    const allChecked = vUsers.length >= 1 && vUsers.every(u => isWebinarChecked(u.id, w.id));
    const item = document.createElement("div");
    item.className = "webinar-item" + (allChecked ? " all-done" : "");

    let usersHtml = "";
    if (vUsers.length === 0) {
      usersHtml = `<p class="no-users-msg">Lägg till superanvändare i steg 1 för att kunna bocka av.</p>`;
    } else {
      usersHtml = vUsers.map(u => {
        const key = u.id + "_" + w.id;
        const checked = isWebinarChecked(u.id, w.id);
        const name = [u.firstName, u.lastName].filter(Boolean).join(" ") || u.email || "Namnlös";
        return `
          <label class="user-check-row">
            <input type="checkbox" ${checked ? "checked" : ""}
              onchange="toggleWebinarCheck('${u.id}','${w.id}')" />
            <span class="user-check-label">
              ${esc(name)}
              ${u.email ? `<br><span class="user-email">${esc(u.email)}</span>` : ""}
            </span>
          </label>`;
      }).join("");
    }

    item.innerHTML = `
      <div class="webinar-head">
        <div class="webinar-meta">
          <span class="webinar-tag">${w.tag}</span>
          <span class="webinar-title">${w.title}</span>
        </div>
        <a href="${w.url}" target="_blank" class="btn-watch">▶ Se webbinar</a>
      </div>
      <div class="webinar-users">${usersHtml}</div>
    `;
    container.appendChild(item);
  });
}

function updateCardStates() {
  // Step 1
  setCardDone("card-users", usersReady());
  setNum("num-users", "1", usersReady());
  // Step 2
  setCardDone("card-webinars", allWebinarsChecked());
  setNum("num-webinars", "2", allWebinarsChecked());
  // Step 3
  setCardDone("card-ansvar", state.ansvarRead);
  setNum("num-ansvar", "3", state.ansvarRead);
  // Step 4
  setCardDone("card-gi", state.giDone);
  setNum("num-gi", "4", state.giDone);
}

function setCardDone(id, done) {
  const el = document.getElementById(id);
  if (!el) return;
  el.classList.toggle("step-done", done);
  el.classList.toggle("active-step", !done);
}
function setNum(id, num, done) {
  const el = document.getElementById(id);
  if (!el) return;
  el.textContent = done ? "✓" : num;
  el.classList.toggle("done", done);
}

function updateCta() {
  const btn = document.getElementById("ctaBtn");
  const sub = document.getElementById("ctaSub");
  const locked = document.getElementById("ctaLocked");
  const remaining = document.getElementById("remainingList");
  const br = document.getElementById("ctaBr");

  if (allDone()) {
    btn.classList.remove("disabled");
    sub.style.display = "";
    locked.style.display = "none";
    remaining.innerHTML = "";
    br.style.display = "";
    btn.href = buildMailto();
  } else {
    btn.classList.add("disabled");
    sub.style.display = "none";
    locked.style.display = "";
    br.style.display = "none";

    const items = [];
    if (!usersReady()) items.push("Registrera minst 2 superanvändare med namn och e-post");
    else if (!allWebinarsChecked()) {
      const vUsers = validUsers();
      WEBINARS.forEach(w => {
        vUsers.forEach(u => {
          if (!isWebinarChecked(u.id, w.id)) {
            const name = [u.firstName, u.lastName].filter(Boolean).join(" ");
            items.push(`${name} – ${w.title}`);
          }
        });
      });
    }
    if (!state.ansvarRead) items.push("Läs och bocka av ansvarsfördelningen");
    if (!state.giDone) items.push("Fyll i och bocka av GI-mallen");

    remaining.innerHTML = items.map(i => `<li>${esc(i)}</li>`).join("");
  }
}

// ── Mailto builder ──────────────────────────────────────
function buildMailto() {
  const vUsers = validUsers();
  const subject = encodeURIComponent(
    `BSO-flytt: ${CONFIG.kundNamn} – Klara för uppstartsmöte`
  );

  let body = `Hej ${CONFIG.konsultNamn},\n\n`;
  body += `Vi på ${CONFIG.kundNamn} har slutfört alla förberedelser inför flytten från BSO till Next.\n\n`;
  body += `── Superanvändare ──────────────────────────\n`;
  vUsers.forEach(u => {
    body += `• ${u.firstName} ${u.lastName}  |  ${u.email}\n`;
  });
  body += `\n── Avklarade steg ──────────────────────────\n`;
  body += `✓ Webbinar 1: Ny i Next\n`;
  body += `✓ Webbinar 2: Grundläggande ekonomiflöde i Next\n`;
  body += `✓ Webbinar 3: Budget, inköp och prognos\n`;
  body += `✓ Ansvarsfördelningen är läst och godkänd\n`;
  body += `✓ GI-mallen är ifylld (se bilaga)\n`;
  body += `\n── Bilaga ──────────────────────────────────\n`;
  body += `Den ifyllda GI-mallen bifogas till detta e-postmeddelande.\n`;
  body += `\nVi ser fram emot att komma igång!\n\nMed vänlig hälsning,\n${CONFIG.kundNamn}`;

  return `mailto:${CONFIG.supportEmail}?subject=${subject}&body=${encodeURIComponent(body)}`;
}

function handleCta(e) {
  if (!allDone()) { e.preventDefault(); return false; }
  return true;
}

// ── Utility ────────────────────────────────────────────
function esc(str) {
  return String(str || "")
    .replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;")
    .replace(/"/g,"&quot;");
}
